01. DBPreceptMessagingPermit.sol

Category: Agent Messaging & State Signal Authorization

Line-by-Line Explanation: Validates messaging permits for autonomous AI agents. The contract verifies an EIP-712 nonce signature against the Clearing House authority before emitting the signal. Single execution ensures zero state pollution. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

/// @title DBPreceptMessagingPermit - Single Action Messaging Guard
/// @notice Zero-fee open source permit under DBPRECEPT Clearing House Standard
contract DBPreceptMessagingPermit {
    address public immutable clearingHouseAdmin;
    mapping(bytes32 => bool) public executedPermits;

    event MessageDispatched(bytes32 indexed permitHash, address indexed sender, string payload);

    constructor() {
        clearingHouseAdmin = msg.sender;
    }

    function dispatchVerifiedMessage(
        string memory payload,
        uint256 nonce,
        bytes memory signature
    ) external {
        bytes32 permitHash = keccak256(abi.encodePacked(msg.sender, payload, nonce));
        require(!executedPermits[permitHash], "DDE: PERMIT_ALREADY_USED");
        
        bytes32 ethSignedMessageHash = keccak256(abi.encodePacked("\x19Ethereum Signed Message:\n32", permitHash));
        address recovered = recoverSigner(ethSignedMessageHash, signature);
        require(recovered == clearingHouseAdmin, "DDE: INVALID_GATEWAY_SIGNATURE");

        executedPermits[permitHash] = true;
        emit MessageDispatched(permitHash, msg.sender, payload);
    }

    function recoverSigner(bytes32 _ethSignedMessageHash, bytes memory _sig) internal pure returns (address) {
        (bytes32 r, bytes32 s, uint8 v) = splitSignature(_sig);
        return ecrecover(_ethSignedMessageHash, v, r, s);
    }

    function splitSignature(bytes memory sig) internal pure returns (bytes32 r, bytes32 s, uint8 v) {
        require(sig.length == 65, "DDE: INVALID_SIG_LENGTH");
        assembly {
            r := mload(add(sig, 32))
            s := mload(add(sig, 64))
            v := byte(0, mload(add(sig, 96)))
        }
    }
}

contract DBPreceptVaultTransferPermit {
    address public immutable vaultOwner;
    uint256 public immutable maxCapPerTransfer;

    event VaultTransferExecuted(address indexed recipient, uint256 amount);

    constructor(uint256 _maxCap) {
        vaultOwner = msg.sender;
        maxCapPerTransfer = _maxCap;
    }

    function executeVaultTransfer(address payable recipient, uint256 amount) external {
        require(msg.sender == vaultOwner, "DDE: UNAUTHORIZED_VAULT_CALLER");
        require(amount <= maxCapPerTransfer, "DDE: EXCEEDS_PERMIT_CAP");
        require(address(this).balance >= amount, "DDE: INSUFFICIENT_VAULT_FUNDS");

        recipient.transfer(amount);
        emit VaultTransferExecuted(recipient, amount);
    }

    receive() external payable {}
}
